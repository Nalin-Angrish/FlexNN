183 return std::visit(

185 using T = std::decay_t<decltype(v)>;
186 if constexpr (std::is_same_v<T, Dense>) {
187 return v.weights().transpose() * dZ;
188 } else if constexpr (std::is_same_v<T, Conv1D>) {
189 return v.propagate(dZ);
190 } else if constexpr (std::is_same_v<T, BatchNorm1D>) {
191 return v.propagate(dZ);
192 } else if constexpr (std::is_same_v<T, MaxPool1D>) {
193 return v.propagate(dZ);
194 } else if constexpr (std::is_same_v<T, AvgPool1D>) {
195 return v.propagate(dZ);
196 } else {
197 return dZ;
198 }

200 var_);

210 std::pair<Eigen::MatrixXd, Eigen::VectorXd> grad(
211 const Eigen::MatrixXd& dZ, const Eigen::MatrixXd& input) const {
212 return std::visit(
213 [&](auto&& v) -> std::pair<Eigen::MatrixXd, Eigen::VectorXd> {
214 using T = std::decay_t<decltype(v)>;
215 if constexpr (std::is_same_v<T, Dense>) {
216 double m = static_cast<double>(dZ.cols());
217 Eigen::MatrixXd dW = dZ * input.transpose() / m;
218 Eigen::VectorXd db = dZ.rowwise().mean();
219 return {dW, db};
220 } else if constexpr (std::is_same_v<T, Conv1D>) {
221 return v.grad(dZ, input);
222 } else if constexpr (std::is_same_v<T, BatchNorm1D>) {
223 return v.grad(dZ, input);
224 } else if constexpr (std::is_same_v<T, MaxPool1D> ||
225 std::is_same_v<T, AvgPool1D>) {
226 return v.grad(dZ, input);
227 } else {
228 return {Eigen::MatrixXd(), Eigen::VectorXd()};
229 }
230 },
231 var_);
232 }
